### 1. Install the dependencies

```bash
pnpm install @tmlmobilidade/ui nuqs swr
```

### 2. Setup `app/layout.tsx`

```tsx
import '@tmlmobilidade/ui/styles.css'; // Import the styles
import { AppProvider } from '@tmlmobilidade/ui'; // Import the AppProvider
import { cookies as nextCookies } from 'next/headers';
import { redirect } from 'next/navigation'; // Import the redirect function
import { Routes } from '@tmlmobilidade/ui/routes'; // Import the Routes object

/* * */

export default async function Layout({ children }) {
	//
	// A. Setup Variables
	const cookies = await nextCookies();
	const sessionToken = cookies.get('session_token')?.value;

	if (!sessionToken) {
		redirect(
			`${Routes.AUTH_API}/login?redirect=${encodeURI(Routes.URL)}`,
			RedirectType.replace,
		);
	}

	//
	// B. Render components
	return (
		<html>
			<body>
				<AppProvider>
					{children}
				</AppProvider>
			</body>
		</html>
	);
}
```

### 3. Setup `lib/routes.ts`

To keep our project clean, we will use a separate file to define our routes. We've referenced the `Routes` object in the `app/layout.tsx` file.

```ts
export const PageRoutes = Object.freeze({
	// ... Application Page Routes
});

export const ApiRoutes = Object.freeze({
	AUTH_API: process.env.NEXT_PUBLIC_AUTH_API_URL,
	AUTH_ME: '/api/users/me',
});

export const Routes = Object.freeze({
	URL: process.env.NEXT_PUBLIC_URL',
	...PageRoutes,
	...ApiRoutes,
});
```
